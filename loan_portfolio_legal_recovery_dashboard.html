// Google Apps Script for Direct 2-Way Sync
var SHEET_ID = '1npUNtFDtBimMH9iXs6Tkk9AWptWPLQgga5R6K4vcFZE';

function getTargetSheet() {
  try {
    return SpreadsheetApp.openById(SHEET_ID).getSheets()[0];
  } catch(e) {
    return SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  }
}

function doGet(e) {
  try {
    var params = e ? e.parameter : {};
    var sheet = getTargetSheet();
    
    // Connection Test
    if (params && params.action === 'test') {
      return ContentService.createTextOutput(JSON.stringify({
        status: 'success',
        sheetName: sheet.getName(),
        rowsCount: sheet.getLastRow(),
        message: 'Connected successfully to Google Sheet!'
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // Direct Update via GET (Bypasses CORS restrictions)
    if (params && params.action === 'update') {
      var result = applyUpdate(sheet, params);
      return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
    }
    
    // Default: Fetch all records for Dashboard
    var data = sheet.getDataRange().getValues();
    return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var sheet = getTargetSheet();
    var payload = {};
    if (e && e.postData && e.postData.contents) {
      try {
        payload = JSON.parse(e.postData.contents);
      } catch(ex) {
        payload = e.parameter || {};
      }
    } else if (e && e.parameter) {
      payload = e.parameter;
    }
    
    var result = applyUpdate(sheet, payload);
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function applyUpdate(sheet, payload) {
  var data = sheet.getDataRange().getValues();
  if (data.length === 0) return { status: 'error', message: 'Sheet is empty' };
  
  // Find header row (check first 5 rows)
  var headerRowIdx = 0;
  for (var r = 0; r < Math.min(data.length, 5); r++) {
    var rowText = data[r].join(' ').toLowerCase();
    if (rowText.indexOf('loan') !== -1 || rowText.indexOf('acc') !== -1 || rowText.indexOf('nic') !== -1) {
      headerRowIdx = r;
      break;
    }
  }
  
  var headers = data[headerRowIdx];
  var accCol = -1, nicCol = -1;
  for (var c = 0; c < headers.length; c++) {
    var h = String(headers[c]).toLowerCase();
    if (/loan|account|acc|facility|contract|a\/c/i.test(h) && accCol === -1) accCol = c;
    if (/nic|identity|id/i.test(h) && nicCol === -1) nicCol = c;
  }
  
  var cleanAcc = String(payload.accountNo || '').replace(/[^a-zA-Z0-9]/g, '').toLowerCase();
  var cleanNic = String(payload.nic || '').replace(/[^a-zA-Z0-9]/g, '').toLowerCase();
  
  var targetRow = -1;
  for (var i = headerRowIdx + 1; i < data.length; i++) {
    var sAcc = accCol !== -1 ? String(data[i][accCol]).replace(/[^a-zA-Z0-9]/g, '').toLowerCase() : '';
    var sNic = nicCol !== -1 ? String(data[i][nicCol]).replace(/[^a-zA-Z0-9]/g, '').toLowerCase() : '';
    
    if ((cleanAcc && sAcc === cleanAcc) || (cleanNic && sNic === cleanNic)) {
      targetRow = i + 1; // 1-based index
      break;
    }
  }
  
  if (targetRow === -1) {
    return { status: 'error', message: 'Account or NIC not found in Sheet: ' + (payload.accountNo || payload.nic) };
  }
  
  // Date Fields mapping
  var fieldMap = {
    'MEDIATION BORD': payload.mediationDate,
    'NON SETELMENT': payload.nonSettlementDate,
    'STETMENT REQ': payload.statementReqDate,
    'LEGALLY': payload.legallyDate,
    'PENDING MEDIATION BORD': payload.pendingMediationDate,
    'PENDING legally': payload.pendingLegallyDate,
    'LOD REQ': payload.lodReqDate
  };
  
  var updatedCols = [];
  for (var col = 0; col < headers.length; col++) {
    var hName = String(headers[col]).trim().toUpperCase();
    for (var key in fieldMap) {
      if (hName.indexOf(key) !== -1 && fieldMap[key] !== undefined && fieldMap[key] !== "") {
        sheet.getRange(targetRow, col + 1).setValue(fieldMap[key]);
        updatedCols.push(key);
      }
    }
  }
  
  return { status: 'success', updatedRow: targetRow, updatedColumns: updatedCols };
}
